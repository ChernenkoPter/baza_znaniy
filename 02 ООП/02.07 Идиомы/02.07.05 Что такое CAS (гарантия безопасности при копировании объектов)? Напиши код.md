Copy-And-Swap - идиома для решения проблем с дублированием кода и со строгой гарантией безопасности исключений (при возникновении исключения объекты остаются в том же состоянии, что до исключения).
```
class MyClass {
    int* data; size_t size;

private:
    void swap(MyClass& other) noexcept {
        using std::swap;
        swap(data, other.data);
        swap(size, other.size);
    }

public:
    MyClass(size_t size) : data(new int[size]()), size(size) {}

    ~MyClass() { delete[] data; }

    MyClass(const MyClass& other) : data(new int[other.size]), size(other.size) {
        std::memcpy(data, other.data, size * sizeof(int));
    }

    // Если копирование упадёт (например, bad_alloc), this не изменится.
    MyClass& operator=(MyClass other) noexcept {
        swap(other); // небросающий обмен ресурсами с копией
        return *this;
    }
};
```
